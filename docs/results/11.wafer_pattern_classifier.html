def generate_sample_wafer_data(
    wafer_diameter_mm=300.0,
    die_size_mm=10.0,
    edge_exclusion_mm=3.0,
    defect_mode="ring"
):
    """
    실습용 웨이퍼 Die 데이터 생성

    wafer_diameter_mm
    → 웨이퍼 직경
    → 300mm 웨이퍼를 기본값으로 사용

    die_size_mm
    → 한 Die의 크기
    → 정사각형 Die라고 단순 가정

    edge_exclusion_mm
    → 웨이퍼 가장자리에서 분석에서 제외할 영역

    defect_mode
    → 불량 패턴 종류
    → ring / center / scratch / random
    """

    # 웨이퍼 반지름 계산
    radius = wafer_diameter_mm / 2.0

    # 가장자리 제외 영역을 반영한 실제 Die 배치 가능 범위
    effective_radius = radius - edge_exclusion_mm

    # Die 중심 좌표 후보 생성
    coords = np.arange(
        -radius,
        radius + die_size_mm,
        die_size_mm
    )

    die_records = []

    # X, Y 좌표를 하나씩 확인하면서
    # 웨이퍼 원 내부에 들어오는 Die만 저장
    for x in coords:
        for y in coords:

            # 웨이퍼 중심에서 현재 Die까지의 거리
            dist = np.sqrt(
                x**2 + y**2
            )

            # Die 전체가 웨이퍼 안쪽에 들어오는 경우만 사용
            if dist + die_size_mm / 2 <= effective_radius:

                die_records.append({
                    'die_x': x,
                    'die_y': y,

                    # 중심으로부터 거리
                    'radius': dist,

                    # 중심 기준 각도
                    'angle_deg': np.degrees(
                        np.arctan2(y, x)
                    )
                })

    # Die 좌표를 DataFrame으로 변환
    df = pd.DataFrame(die_records)

        # --------------------------------
    # 불량 패턴 적용
    # --------------------------------

    if defect_mode == 'ring':

        # Ring 불량
        # 웨이퍼 가장자리 특정 반경 구간을 FAIL 처리
        df['status'] = np.where(
            (df['radius'] >= 120) &
            (df['radius'] <= 145),
            'FAIL',
            'PASS'
        )

    elif defect_mode == 'center':

        # Center 불량
        # 웨이퍼 중심 30mm 이내를 FAIL 처리
        df['status'] = np.where(
            df['radius'] <= 30,
            'FAIL',
            'PASS'
        )

    elif defect_mode == 'scratch':

        # Scratch 불량
        # 대각선 방향의 불량 패턴을 단순 가정
        df['status'] = np.where(
            (df['die_x'] + df['die_y']).between(-15, 15),
            'FAIL',
            'PASS'
        )

    else:

        # Random 불량
        # 전체 Die 중 약 5%를 무작위 FAIL 처리
        np.random.seed(42)

        df['status'] = np.where(
            np.random.random(len(df)) < 0.05,
            'FAIL',
            'PASS'
        )

    # --------------------------------
    # Yield 계산
    # --------------------------------

    # 전체 Die 개수
    total = len(df)

    # PASS Die 개수
    passed = (df['status'] == 'PASS').sum()

    # Yield 계산
    yield_pct = passed / total * 100

    # DataFrame에 추가 정보 저장
    df.attrs['yield_pct'] = yield_pct
    df.attrs['total_dies'] = total
    df.attrs['defect_mode'] = defect_mode

    # 완성된 웨이퍼 데이터 반환
    return df

def plot_wafer_map(
    df,
    lot_id="LOT_001",
    wafer_id="W01",
    die_size_mm=10.0,
    wafer_radius_mm=150.0
):
    """
    생성된 웨이퍼 Die 데이터를
    PASS / FAIL 웨이퍼 맵으로 시각화
    """

    # 그래프 생성
    fig, ax = plt.subplots(figsize=(8, 8))

    # PASS / FAIL 색상
    color_map = {
        "PASS": "#4CAF50",
        "FAIL": "#F44336"
    }

    # Die 크기의 절반
    half = die_size_mm / 2.0

    # --------------------------------
    # 각 Die를 사각형으로 표시
    # --------------------------------

    for _, row in df.iterrows():

        color = color_map.get(
            row["status"],
            "#9E9E9E"
        )

        rect = plt.Rectangle(
            (
                row["die_x"] - half,
                row["die_y"] - half
            ),
            die_size_mm * 0.9,
            die_size_mm * 0.9,
            linewidth=0.3,
            edgecolor="white",
            facecolor=color,
            alpha=0.85
        )

        ax.add_patch(rect)

    # --------------------------------
    # 웨이퍼 외곽 원
    # --------------------------------

    wafer_circle = Circle(
        (0, 0),
        wafer_radius_mm,
        fill=False,
        linewidth=2
    )

    ax.add_patch(wafer_circle)

    # --------------------------------
    # 그래프 범위 설정
    # --------------------------------

    ax.set_xlim(
        -wafer_radius_mm * 1.1,
        wafer_radius_mm * 1.1
    )

    ax.set_ylim(
        -wafer_radius_mm * 1.1,
        wafer_radius_mm * 1.1
    )

    # 웨이퍼가 찌그러지지 않도록
    # X/Y 비율 동일하게 설정
    ax.set_aspect("equal")

    ax.set_xlabel("X Position (mm)")
    ax.set_ylabel("Y Position (mm)")

    # --------------------------------
    # Yield 정보
    # --------------------------------

    yield_pct = df.attrs["yield_pct"]
    total_dies = df.attrs["total_dies"]
    fail_dies = (df["status"] == "FAIL").sum()

    ax.set_title(
        f"Lot: {lot_id} | Wafer: {wafer_id}\n"
        f"Yield: {yield_pct:.2f}% | "
        f"Total: {total_dies} | "
        f"FAIL: {fail_dies}"
    )

    # --------------------------------
    # PASS / FAIL 범례
    # --------------------------------

    legend_elements = [
        mpatches.Patch(
            facecolor="#4CAF50",
            label="PASS"
        ),
        mpatches.Patch(
            facecolor="#F44336",
            label="FAIL"
        )
    ]

    ax.legend(
        handles=legend_elements,
        loc="lower center"
    )

    plt.tight_layout()

    return fig, ax

class WaferPatternResult:

    # 자동 분류된 불량 패턴

    # 패턴에 따른 권고사항

# 13. 웨이퍼 불량 패턴 자동 분류 함수
# --------------------------------

def classify_wafer_pattern(
    df: pd.DataFrame,
    wafer_radius: float = 150.0
) -> WaferPatternResult:

    """
    웨이퍼의 FAIL Die 위치를 이용해
    불량 패턴을 자동으로 분류한다.

    분류 대상
    ----------
    CLEAN
    EDGE_RING
    CENTER
    SCRATCH
    RANDOM
    """

    # --------------------------------
    # FAIL Die만 선택
    # --------------------------------

    fail_df = df[
        df["status"] == "FAIL"
    ].copy()

    # 전체 Die 개수
    total = len(df)

    # FAIL Die 개수
    fail_count = len(fail_df)

    # Yield 계산
    yield_pct = (
        (total - fail_count)
        / total
        * 100
    )

    # --------------------------------
    # CLEAN 판정
    # --------------------------------

    # 불량이 하나도 없거나
    # Yield가 98%를 초과하면 CLEAN으로 분류

    if fail_count == 0 or yield_pct > 98.0:

        return WaferPatternResult(
            pattern="CLEAN",
            confidence=1.0,
            yield_pct=yield_pct,
            fail_count=fail_count,
            fail_radius_mean=0.0,
            fail_radius_std=0.0,
            spatial_autocorr=0.0,
            linearity=0.0,
            recommended_action=(
                "특별한 불량 패턴이 감지되지 않았습니다. "
                "공정 추세를 지속적으로 모니터링합니다."
            )
        )

    # --------------------------------
    # 불량 반경 특성 계산
    # --------------------------------

    fail_radius_mean = (
        fail_df["radius"].mean()
    )

    fail_radius_std = (
        fail_df["radius"].std()
    )

    # 표준편차가 NaN인 경우 0으로 처리
    if pd.isna(fail_radius_std):
        fail_radius_std = 0.0

    # --------------------------------
    # 공간 집중도 계산
    # --------------------------------

    # FAIL Die 사이의 최근접 거리를 이용해
    # 불량들이 서로 얼마나 가까이 모여 있는지 단순 계산

    if fail_count > 1:

        x_vals = (
            fail_df["die_x"]
            .to_numpy()
        )

        y_vals = (
            fail_df["die_y"]
            .to_numpy()
        )

        # 모든 FAIL Die 사이 거리 계산
        dist_matrix = np.sqrt(
            (
                x_vals[:, None]
                - x_vals[None, :]
            ) ** 2
            +
            (
                y_vals[:, None]
                - y_vals[None, :]
            ) ** 2
        )

        # 자기 자신과의 거리는 제외
        np.fill_diagonal(
            dist_matrix,
            np.inf
        )

        # 각 FAIL Die에서 가장 가까운
        # 다른 FAIL Die까지의 거리
        min_dist = (
            dist_matrix.min(axis=1)
        )

        # 값이 1에 가까울수록
        # 불량들이 가까이 모여 있다는 의미
        spatial_autocorr = float(
            np.clip(
                1
                - (
                    min_dist.mean()
                    / wafer_radius
                ),
                0,
                1
            )
        )

    else:

        spatial_autocorr = 0.0

    # --------------------------------
    # Scratch 방향성 계산
    # --------------------------------

    # FAIL Die들이 한 줄 형태로 늘어서 있는지
    # PCA 방식으로 선형성을 계산
    #
    # linearity → 1에 가까울수록
    # 한 방향으로 길게 늘어진 패턴

    if fail_count >= 3:

        coords = fail_df[
            ["die_x", "die_y"]
        ].to_numpy()

        # 좌표 중심 맞추기
        coords_centered = (
            coords
            - coords.mean(axis=0)
        )

        # 공분산 행렬
        cov_matrix = np.cov(
            coords_centered,
            rowvar=False
        )

        # 고유값 계산
        eigenvalues = np.linalg.eigvalsh(
            cov_matrix
        )

        # 작은 값 → 큰 값 순으로 정렬
        eigenvalues = np.sort(
            eigenvalues
        )

        # 가장 큰 방향의 분산이
        # 전체 분산에서 차지하는 정도를 이용
        if eigenvalues[-1] > 0:

            linearity = (
                1
                - (
                    eigenvalues[0]
                    / eigenvalues[-1]
                )
            )

        else:

            linearity = 0.0

    else:

        linearity = 0.0

    # --------------------------------
    # 패턴 분류
    # --------------------------------

    # 1. EDGE RING
    # FAIL Die들이 웨이퍼 외곽에 몰려 있고
    # 반경 분포가 비교적 좁은 경우

    if (
        fail_radius_mean
        > wafer_radius * 0.75
        and fail_radius_std < 20
    ):

        pattern = "EDGE_RING"

        confidence = np.clip(
            fail_radius_mean
            / wafer_radius,
            0,
            1
        )

        action = (
            "엣지 링 불량 감지. "
            "증착 균일도, Edge 영역 및 "
            "공정 조건 점검을 권장합니다."
        )

    # --------------------------------
    # 2. CENTER
    # --------------------------------

    elif (
        fail_radius_mean
        < wafer_radius * 0.25
    ):

        pattern = "CENTER"

        confidence = np.clip(
            1
            - (
                fail_radius_mean
                / (wafer_radius * 0.25)
            ),
            0,
            1
        )

        action = (
            "중심부 불량 감지. "
            "Chuck 온도 분포, 중심부 가스 흐름 등 "
            "Center 관련 공정 조건 점검을 권장합니다."
        )

    # --------------------------------
    # 3. SCRATCH
    # --------------------------------

    elif linearity > 0.85:

        pattern = "SCRATCH"

        confidence = float(
            np.clip(
                linearity,
                0,
                1
            )
        )

        action = (
            "선형 Scratch 패턴 감지. "
            "CMP Pad 상태, Wafer Handling 및 "
            "파티클 오염 여부 점검을 권장합니다."
        )

    # --------------------------------
    # 4. RANDOM
    # --------------------------------

    else:

        pattern = "RANDOM"

        confidence = float(
            np.clip(
                1 - linearity,
                0,
                1
            )
        )

        action = (
            "뚜렷한 공간 패턴이 없는 Random 불량입니다. "
            "Particle Count, 설비 이력 및 "
            "공정 변동 추세 확인을 권장합니다."
        )

    # --------------------------------
    # 결과 반환
    # --------------------------------

    return WaferPatternResult(
        pattern=pattern,
        confidence=float(confidence),
        yield_pct=float(yield_pct),
        fail_count=int(fail_count),
        fail_radius_mean=float(
            fail_radius_mean
        ),
        fail_radius_std=float(
            fail_radius_std
        ),
        spatial_autocorr=float(
            spatial_autocorr
        ),
        linearity=float(
            linearity
        ),
        recommended_action=action
    )
